from PIL import Image,ImageDraw,ImageFont
base='docs/requirements/2026-10-28-view-modes-seat-camera/evidence'

for i,(n,im) in enumerate(imgs):
    x=(i%2)*720; y=(i//2)*450; sheet.paste(im,(x+10,y+40))
    label=f'{n} player(s)' + ('  waiting (min 2)' if n==1 else f'  models={n}/{n}')
    d.text((x+12,y+8),label,fill=(240,220,180),font=font)
sheet.save(f'{base}/online-1-to-8-grid.jpg',quality=90)
